{{/* Venue line: journal / book / review name only. Volume, issue and pages are deliberately not shown here; they stay in the front matter and are used for BibTeX (cite.bib) and the Google Scholar meta tags. */}} {{- $pub := .Params.publication | default dict -}} {{- $type := index (.Params.publication_types | default (slice "")) 0 -}} {{- $out := "" -}} {{- if reflect.IsMap $pub -}} {{- if eq $type "chapter" -}} {{- $out = printf "In %s" $pub.name -}} {{- with $pub.editor }}{{ $out = printf "%s, ed. %s" $out . }}{{ end -}} {{- with $pub.publisher }}{{ $out = printf "%s. %s" $out . }}{{ end -}} {{- else if eq $type "report" -}} {{- $out = printf "%s" $pub.name -}} {{- with $pub.publisher }}{{ $out = printf "%s, %s" $out . }}{{ end -}} {{- else -}} {{- $out = printf "%s" $pub.name -}} {{- with .Params.publication_note }}{{ $out = printf "%s in %s" . $out }}{{ end -}} {{- end -}} {{- end -}} {{- $out | safeHTML -}}
